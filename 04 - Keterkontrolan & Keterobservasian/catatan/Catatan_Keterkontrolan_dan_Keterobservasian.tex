\section{Pendahuluan \& Konsep Struktural Sistem}

Konsep \textbf{Keterkontrolan (\textit{Controllability})} dan \textbf{Keterobservasian (\textit{Observability})} diperkenalkan oleh Rudolf E. Kalman pada awal 1960-an sebagai pilar fundamental teori kontrol modern berbasis ruang keadaan (\textit{state-space}):

\begin{itemize}
    \item \textbf{Keterkontrolan}: Menjawab apakah masukan kendali $u(t)$ mampu mengarahkan seluruh variabel keadaan internal $x(t)$ dari sembarang kondisi awal menuju titik tujuan dalam waktu berhingga.
    \item \textbf{Keterobservasian}: Menjawab apakah kondisi awal seluruh variabel keadaan internal $x(0)$ dapat direkonstruksi secara unik hanya melalui pengukuran riwayat keluaran $y(t)$ dan masukan $u(t)$.
\end{itemize}

\vspace{6pt}

\section{Keterkontrolan Keadaan (\textit{State Controllability})}

Diberikan sistem LTI kontinu:
\begin{equation*}
    \dot{x}(t) = A x(t) + B u(t), \quad x(t) \in \mathbb{R}^n, \; u(t) \in \mathbb{R}^m
\end{equation*}

\kotakdefinisi{Teorema Keterkontrolan Kalman:}{
Sistem kontinu LTI $(A, B)$ dengan $n$ variabel keadaan bersifat \textbf{terkontrol keadaan lengkap} jika dan hanya jika matriks keterkontrolan $M_c$ memiliki rank penuh:
\begin{equation*}
    \operatorname{rank}(M_c) = \operatorname{rank} \begin{bmatrix} B & AB & A^2 B & \dots & A^{n-1} B \end{bmatrix} = n
\end{equation*}
Untuk sistem SISO ($m = 1$), syarat ekuivalennya adalah $\det(M_c) \ne 0$.
}

\subsection{Uji Popov-Belevitch-Hautus (PBH Test) Keterkontrolan}
\kotakdefinisi{Uji PBH Keterkontrolan:}{
Sistem $(A, B)$ terkontrol lengkap jika dan hanya jika:
\begin{equation*}
    \operatorname{rank} \begin{bmatrix} sI - A & B \end{bmatrix} = n, \quad \forall s \in \mathbb{C}
\end{equation*}
Cukup diuji pada seluruh nilai eigen $s = \lambda_i$ dari matriks $A$.
}

\subsection{Gramian Keterkontrolan (\textit{Controllability Gramian})}
\begin{equation*}
    W_c(0, t_1) = \int_0^{t_1} e^{A(t_1 - \tau)} B B^T e^{A^T(t_1 - \tau)} \, d\tau > 0 \quad (\forall t_1 > 0)
\end{equation*}

\vspace{6pt}

\section{Keterobservasian Keadaan (\textit{State Observability})}

Diberikan sistem LTI kontinu:
\begin{equation*}
    \dot{x}(t) = A x(t) + B u(t), \quad y(t) = C x(t) + D u(t)
\end{equation*}

\kotakdefinisi{Teorema Keterobservasian Kalman:}{
Sistem kontinu LTI $(A, C)$ bersifat \textbf{terobservasi keadaan lengkap} jika dan hanya jika matriks keterobservasian $M_o$ memiliki rank penuh:
\begin{equation*}
    \operatorname{rank}(M_o) = \operatorname{rank} \begin{bmatrix} C \\ CA \\ CA^2 \\ \vdots \\ CA^{n-1} \end{bmatrix} = n
\end{equation*}
Untuk sistem keluaran skalar ($p = 1$), syarat ekuivalennya adalah $\det(M_o) \ne 0$.
}

\subsection{Uji PBH Keterobservasian}
\kotakdefinisi{Uji PBH Keterobservasian:}{
Sistem $(A, C)$ terobservasi lengkap jika dan hanya jika:
\begin{equation*}
    \operatorname{rank} \begin{bmatrix} sI - A \\ C \end{bmatrix} = n, \quad \forall s \in \mathbb{C}
\end{equation*}
Cukup diuji pada seluruh nilai eigen $s = \lambda_i$ dari matriks $A$.
}

\vspace{6pt}

\section{Prinsip Dualitas Kalman}

\kotakdefinisi{Teorema Dualitas Kalman:}{
Pasangan sistem $(A, B)$ bersifat \textbf{terkontrol lengkap} jika dan hanya jika pasangan dualnya $(A^T, B^T)$ bersifat \textbf{terobservasi lengkap}.
\begin{equation*}
    M_c(A, B) = M_o^T(A^T, B^T)
\end{equation*}
}

\vspace{6pt}

\section{Matriks Transformasi ke Bentuk Kanonik (\texorpdfstring{$P$}{P} dan \texorpdfstring{$Q$}{Q})}

Misalkan polinomial karakteristik $\det(sI - A) = s^n + a_1 s^{n-1} + \dots + a_n$.
Definisikan matriks Toeplitz $W$:
\begin{equation*}
    W = \begin{bmatrix}
        a_{n-1} & a_{n-2} & \dots & a_1 & 1 \\
        a_{n-2} & a_{n-3} & \dots & 1 & 0 \\
        \vdots & \vdots & \ddots & \vdots & \vdots \\
        a_1 & 1 & \dots & 0 & 0 \\
        1 & 0 & \dots & 0 & 0
    \end{bmatrix}
\end{equation*}

\kotakdefinisi{Matriks Transformasi Keterkontrolan:}{
Matriks transformasi $T = M_c W$ mentransformasikan sistem $(A, B)$ menjadi bentuk kanonik terkontrol:
\begin{equation*}
    T^{-1} A T = \begin{bmatrix}
        0 & 1 & 0 & \dots & 0 \\
        0 & 0 & 1 & \dots & 0 \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & 0 & \dots & 1 \\
        -a_n & -a_{n-1} & -a_{n-2} & \dots & -a_1
    \end{bmatrix}, \quad
    T^{-1} B = \begin{bmatrix} 0 \\ 0 \\ \vdots \\ 0 \\ 1 \end{bmatrix}
\end{equation*}
}

\vspace{6pt}

\section{Keterkontrolan Keluaran (\textit{Output Controllability})}

\kotakdefinisi{Syarat Rank Keterkontrolan Keluaran:}{
Sistem $\dot{x} = Ax + Bu, y = Cx + Du$ dengan $p$ keluaran terkontrol keluaran lengkap jika dan hanya jika:
\begin{equation*}
    \operatorname{rank}(M_{co}) = \operatorname{rank} \begin{bmatrix} CB & CAB & CA^2 B & \dots & CA^{n-1} B & D \end{bmatrix} = p
\end{equation*}
}

\vspace{6pt}

\section{Keterstabilan (\textit{Stabilizability}) \& Keterdeteksian (\textit{Detectability})}

\begin{align*}
    (A, B) \text{ stabilizable} &\iff \operatorname{rank} \begin{bmatrix} sI - A & B \end{bmatrix} = n, \quad \forall s \in \mathbb{C} \text{ dengan } \operatorname{Re}(s) \ge 0 \\
    (A, C) \text{ detectable} &\iff \operatorname{rank} \begin{bmatrix} sI - A \\ C \end{bmatrix} = n, \quad \forall s \in \mathbb{C} \text{ dengan } \operatorname{Re}(s) \ge 0
\end{align*}

\vspace{6pt}

\section{Dekomposisi Kanonik Kalman \& Pembatalan Kutub-Nol}

\kotakdefinisi{Teorema Realisasi Minimal \& Pembatalan Kutub-Nol:}{
\begin{enumerate}
    \item Representasi ruang keadaan $(A, B, C, D)$ berorde $n$ merupakan \textbf{realisasi minimal} dari fungsi transfer $G(s)$ jika dan hanya jika sistem bersifat \textbf{terkontrol lengkap dan terobservasi lengkap}.
    \item Jika terjadi \textbf{pembatalan kutub-pembuat nol (\textit{pole-zero cancellation})} pada $G(s)$, maka representasi ruang keadaan pasti kehilangan sifat keterkontrolan, keterobservasian, atau keduanya.
\end{enumerate}
}

\vspace{6pt}

\section{Keterkontrolan \& Keterobservasian Sistem Diskrit}

Untuk sistem diskrit $x(k+1) = G x(k) + H u(k), \; y(k) = C x(k) + D u(k)$:
\begin{align*}
    M_c &= \begin{bmatrix} H & GH & G^2 H & \dots & G^{n-1} H \end{bmatrix}, \quad \operatorname{rank}(M_c) = n \\
    M_o &= \begin{bmatrix} C \\ CG \\ CG^2 \\ \vdots \\ CG^{n-1} \end{bmatrix}, \quad \operatorname{rank}(M_o) = n
\end{align*}

\vspace{6pt}

\section{Contoh Soal dan Pembahasan Komprehensif}

\noindent\textbf{Contoh 1: Uji Keterkontrolan dan Keterobservasian Berparameter}\\
Diberikan sistem:
\begin{equation*}
    \dot{x} = \begin{bmatrix} 1 & 2 \\ -4 & -3 \end{bmatrix} x + \begin{bmatrix} 1 \\ 2 \end{bmatrix} u, \quad y = \begin{bmatrix} a & 1 \end{bmatrix} x
\end{equation*}

\textit{Penyelesaian:}

1. \textbf{Keterkontrolan}:
\begin{equation*}
    AB = \begin{bmatrix} 1 & 2 \\ -4 & -3 \end{bmatrix} \begin{bmatrix} 1 \\ 2 \end{bmatrix} = \begin{bmatrix} 5 \\ -10 \end{bmatrix} \implies M_c = \begin{bmatrix} 1 & 5 \\ 2 & -10 \end{bmatrix}
\end{equation*}
$\det(M_c) = -10 - 10 = -20 \ne 0 \implies \operatorname{rank}(M_c) = 2$. Sistem \textbf{terkontrol keadaan lengkap}.

2. \textbf{Keterobservasian}:
\begin{equation*}
    CA = \begin{bmatrix} a & 1 \end{bmatrix} \begin{bmatrix} 1 & 2 \\ -4 & -3 \end{bmatrix} = \begin{bmatrix} a - 4 & 2a - 3 \end{bmatrix} \implies M_o = \begin{bmatrix} a & 1 \\ a - 4 & 2a - 3 \end{bmatrix}
\end{equation*}
Determinan:
\begin{equation*}
    \det(M_o) = a(2a - 3) - 1(a - 4) = 2a^2 - 4a + 4 = 2[(a-1)^2 + 1]
\end{equation*}
Karena $(a-1)^2 + 1 > 0$ untuk setiap $a \in \mathbb{R}$, maka $\det(M_o) \ne 0$ untuk seluruh $a \in \mathbb{R}$.  
Sistem \textbf{terobservasi keadaan lengkap untuk semua $a \in \mathbb{R}$}.

\vspace{10pt}

\noindent\textbf{Contoh 2: Analisis Pembatalan Pole-Zero}\\
Diberikan $G(s) = \dfrac{s + 1}{(s + 1)(s + 2)} = \dfrac{s + 1}{s^2 + 3s + 2}$. Bentuk kanonik terkontrol:
\begin{equation*}
    A = \begin{bmatrix} 0 & 1 \\ -2 & -3 \end{bmatrix}, \quad B = \begin{bmatrix} 0 \\ 1 \end{bmatrix}, \quad C = \begin{bmatrix} 1 & 1 \end{bmatrix}
\end{equation*}
Diperoleh:
\begin{itemize}
    \item $M_c = \begin{bmatrix} 0 & 1 \\ 1 & -3 \end{bmatrix} \implies \det(M_c) = -1 \ne 0 \implies \operatorname{rank}(M_c) = 2$ (\textbf{Terkontrol}).
    \item $M_o = \begin{bmatrix} 1 & 1 \\ -2 & -2 \end{bmatrix} \implies \det(M_o) = 0 \implies \operatorname{rank}(M_o) = 1 < 2$ (\textbf{Tak Terobservasi}).
\end{itemize}
Pembatalan kutub $(s + 1)$ membuat mode internal $\lambda = -1$ tak terobservasi.
