\section{Pendahuluan \& Karakteristik Sistem LTI}

Pemodelan matematika merupakan fondasi dasar dalam menganalisis perilaku sistem dinamik dan merancang pengontrol yang optimal, stabil, dan tegar (\textit{robust}).

\subsection{Karakteristik Sistem Linear Time-Invariant (LTI)}
Kajian utama mata kuliah ini berfokus pada sistem \textbf{Linear Time-Invariant (LTI)} kontinu:
\begin{enumerate}
    \item \textbf{Prinsip Superposisi \& Homogenitas (Linearitas)}:\\
    Jika masukan $u_1(t)$ menghasilkan keluaran $y_1(t)$ dan $u_2(t)$ menghasilkan $y_2(t)$, maka untuk setiap skalar $\alpha, \beta \in \mathbb{R}$:
    \begin{equation*}
        \alpha u_1(t) + \beta u_2(t) \implies \alpha y_1(t) + \beta y_2(t)
    \end{equation*}
    \item \textbf{Time-Invariance (Invarian Waktu)}:\\
    Karakteristik parameter sistem tidak bergantung pada waktu pergeseran. Jika $u(t) \to y(t)$, maka untuk sembarang pergeseran $\tau \ge 0$:
    \begin{equation*}
        u(t - \tau) \implies y(t - \tau)
    \end{equation*}
\end{enumerate}

\subsection{Dua Pendekatan Pemodelan Sistem LTI}
\begin{itemize}
    \item \textbf{Domain Frekuensi (S-Domain / Kontrol Klasik)}: Berbasis \textit{Fungsi Transfer}, \textit{Diagram Blok}, dan \textit{Signal Flow Graph}. Pendekatan ini sangat efektif untuk sistem SISO (\textit{Single-Input Single-Output}) dengan asumsi kondisi awal nol.
    \item \textbf{Domain Waktu (State-Space / Kontrol Modern)}: Berbasis sistem persamaan diferensial vektor orde satu simultan. Pendekatan ini mampu menangani sistem multivariabel MIMO (\textit{Multi-Input Multi-Output}) dan kondisi awal riil.
\end{itemize}

\vspace{6pt}

\section{Fungsi Transfer (\textit{Transfer Function})}

\kotakdefinisi{Definisi Fungsi Transfer:}{
Fungsi transfer sistem LTI didefinisikan sebagai rasio antara Transformasi Laplace dari sinyal keluaran $Y(s)$ terhadap Transformasi Laplace dari sinyal masukan $U(s)$ dengan \textbf{seluruh kondisi awal bernilai nol}:
\begin{equation*}
    G(s) = \frac{Y(s)}{U(s)} = \frac{\mathcal{L}\{y(t)\}}{\mathcal{L}\{u(t)\}}\Bigg|_{\text{seluruh kondisi awal } = 0}
\end{equation*}
}

Jika sistem fisik dimodelkan oleh persamaan diferensial linear orde-$n$:
\begin{equation*}
    a_0 \frac{d^n y}{dt^n} + a_1 \frac{d^{n-1} y}{dt^{n-1}} + \dots + a_{n-1} \frac{dy}{dt} + a_n y(t) = b_0 \frac{d^m u}{dt^m} + b_1 \frac{d^{m-1} u}{dt^{m-1}} + \dots + b_m u(t) \quad (n \ge m)
\end{equation*}
Dengan menerapkan Transformasi Laplace dan menyetel seluruh turunan awal ke nol:
\begin{equation*}
    (a_0 s^n + a_1 s^{n-1} + \dots + a_n) Y(s) = (b_0 s^m + b_1 s^{m-1} + \dots + b_m) U(s)
\end{equation*}
Diperoleh bentuk umum fungsi transfer:
\begin{equation*}
    G(s) = \frac{N(s)}{D(s)} = \frac{b_0 s^m + b_1 s^{m-1} + \dots + b_{m-1} s + b_m}{a_0 s^n + a_1 s^{n-1} + \dots + a_{n-1} s + a_n}
\end{equation*}

\subsection{Karakteristik Penting Fungsi Transfer}
\begin{itemize}
    \item \textbf{Persamaan Karakteristik}: Polinomial penyebut yang disamakan dengan nol, yaitu $D(s) = 0$.
    \item \textbf{Kutub (\textit{Poles})}: Nilai-nilai $s$ yang menyebabkan $D(s) = 0$ ($G(s) \to \infty$). Letak kutub pada bidang kompleks $s$ menentukan kestabilan dan respon alami sistem.
    \item \textbf{Pembuat Nol (\textit{Zeros})}: Nilai-nilai $s$ yang menyebabkan $N(s) = 0$ ($G(s) = 0$).
    \item \textbf{Tanggapan Impuls (\textit{Impulse Response})}: Respon sistem terhadap masukan Dirac delta $\delta(t)$ ($\mathcal{L}\{\delta(t)\} = 1$):
    \begin{equation*}
        y(t) = g(t) = \mathcal{L}^{-1}\{G(s)\}
    \end{equation*}
\end{itemize}

\section{Aljabar dan Reduksi Diagram Blok}

Diagram blok menyajikan interkoneksi fungsional antar subsistem.

\subsection{Operasi Ekuivalensi Diagram Blok}
\begin{center}
\small
\begin{tabular}{|p{3.5cm}|p{5cm}|p{6cm}|}
\hline
\textbf{Konfigurasi} & \textbf{Deskripsi Hubungan} & \textbf{Fungsi Transfer Ekuivalen} \\ \hline
\textbf{Kaskade (Seri)} & Sinyal mengalir berurutan & $G_{eq}(s) = G_1(s) \cdot G_2(s)$ \\ \hline
\textbf{Paralel} & Sinyal masukan sama dijumlahkan & $G_{eq}(s) = G_1(s) \pm G_2(s)$ \\ \hline
\textbf{Loop Umpan Balik Negatif} & Forward $G(s)$, Feedback $H(s)$ (-) & $T(s) = \dfrac{Y(s)}{R(s)} = \dfrac{G(s)}{1 + G(s)H(s)}$ \\ \hline
\textbf{Loop Umpan Balik Positif} & Forward $G(s)$, Feedback $H(s)$ (+) & $T(s) = \dfrac{Y(s)}{R(s)} = \dfrac{G(s)}{1 - G(s)H(s)}$ \\ \hline
\textbf{Pindah Titik Cabang} & Maju melewati blok $G(s)$ & Jalur cabang dikalikan $\dfrac{1}{G(s)}$ \\ \hline
\textbf{Pindah Titik Jumlah} & Maju melewati blok $G(s)$ & Jalur penjumlahan dikalikan $G(s)$ \\ \hline
\end{tabular}
\end{center}

Untuk sistem dengan gangguan (\textit{disturbance}) $D(s)$ dan masukan acuan $R(s)$, keluaran dihitung menggunakan prinsip superposisi:
\begin{equation*}
    C(s) = T_R(s) R(s) + T_D(s) D(s)
\end{equation*}

\section{Signal-Flow Graph (SFG) \& Aturan Penguatan Mason}

\textbf{Signal-Flow Graph (SFG)} adalah representasi graf berarah dari sistem persamaan aljabar linear simultan.

\subsection{Terminologi Graf Aliran Sinyal}
\begin{itemize}
    \item \textbf{Simpul Masukan (\textit{Source Node})}: Simpul yang hanya memiliki cabang keluar (contoh: $R(s)$).
    \item \textbf{Simpul Keluaran (\textit{Sink Node})}: Simpul yang hanya memiliki cabang masuk (contoh: $C(s)$).
    \item \textbf{Transmitansi (\textit{Gain})}: Bobot pengali sinyal pada suatu cabang.
    \item \textbf{Jalur Maju (\textit{Forward Path}, $F_k$)}: Lintasan dari simpul masukan ke simpul keluaran yang searah dengan panah tanpa melewati simpul yang sama lebih dari satu kali.
    \item \textbf{Lup (\textit{Loop}, $L_i$)}: Lintasan tertutup yang dimulai dan berakhir pada simpul yang sama tanpa melintasi simpul lain lebih dari sekali.
    \item \textbf{Lup Tak Bersentuhan (\textit{Non-touching Loops})}: Lup-lup yang tidak berbagi simpul yang sama.
\end{itemize}

\subsection{Rumus Penguatan Mason (\textit{Mason's Gain Formula})}
\kotakdefinisi{Aturan Mason:}{
Fungsi transfer total dari sinyal masukan ke keluaran dihitung melalui:
\begin{equation*}
    G(s) = \frac{C(s)}{R(s)} = \frac{\sum_{k=1}^N F_k \Delta_k}{\Delta}
\end{equation*}
Di mana:
\begin{itemize}
    \item $N$ = Jumlah total lintasan maju (\textit{forward paths}).
    \item $F_k$ = Penguatan lintasan maju ke-$k$.
    \item $\Delta$ = Determinan sistem graf:
    \begin{equation*}
        \Delta = 1 - \sum L_i + \sum L_i L_j - \sum L_i L_j L_k + \dots
    \end{equation*}
    \item $\Delta_k$ = Kofaktor dari lintasan maju ke-$k$, yaitu nilai determinan $\Delta$ yang dihitung hanya untuk bagian graf yang \textbf{tidak menyentuh} lintasan maju ke-$k$.
\end{itemize}
}

\section{Representasi Ruang Keadaan (\textit{State-Space Representation})}

\subsection{Bentuk Standar Persamaan Ruang Keadaan LTI}
\kotakdefinisi{Sistem Ruang Keadaan LTI Kontinu:}{
\begin{align*}
    \dot{x}(t) &= A x(t) + B u(t) \quad &\text{(\textbf{Persamaan Keadaan} / \textit{State Equation})} \\
    y(t) &= C x(t) + D u(t) \quad &\text{(\textbf{Persamaan Keluaran} / \textit{Output Equation})}
\end{align*}
}

Dimensi matriks untuk sistem dengan $n$ variabel keadaan, $m$ masukan, dan $p$ keluaran:
\begin{itemize}
    \item $A \in \mathbb{R}^{n \times n}$ : Matriks dinamika sistem.
    \item $B \in \mathbb{R}^{n \times m}$ : Matriks masukan.
    \item $C \in \mathbb{R}^{p \times n}$ : Matriks keluaran.
    \item $D \in \mathbb{R}^{p \times m}$ : Matriks transmisi langsung.
\end{itemize}

\vspace{6pt}

\section{Hubungan Ruang Keadaan dan Fungsi Transfer}

Dengan menerapkan Transformasi Laplace pada persamaan keadaan dengan syarat awal $x(0) = 0$:
\begin{align*}
    s X(s) - x(0) &= A X(s) + B U(s) \implies (sI - A) X(s) = B U(s) \implies X(s) = (sI - A)^{-1} B U(s)
\end{align*}
Substitusikan ke persamaan keluaran $Y(s) = C X(s) + D U(s)$:
\begin{equation*}
    G(s) = \frac{Y(s)}{U(s)} = C(sI - A)^{-1}B + D = \frac{C \operatorname{adj}(sI - A) B}{\det(sI - A)} + D
\end{equation*}

\vspace{6pt}

\section{Bentuk-Bentuk Kanonik Ruang Keadaan}

Diberikan fungsi transfer SISO \textit{strictly proper}:
\begin{equation*}
    G(s) = \frac{b_1 s^{n-1} + b_2 s^{n-2} + \dots + b_n}{s^n + a_1 s^{n-1} + a_2 s^{n-2} + \dots + a_n}
\end{equation*}

\subsection{1. Bentuk Kanonik Keterkontrolan (\textit{Controllable Canonical Form})}
\begin{equation*}
    A = \begin{bmatrix}
        0 & 1 & 0 & \dots & 0 \\
        0 & 0 & 1 & \dots & 0 \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & 0 & \dots & 1 \\
        -a_n & -a_{n-1} & -a_{n-2} & \dots & -a_1
    \end{bmatrix}, \quad
    B = \begin{bmatrix} 0 \\ 0 \\ \vdots \\ 0 \\ 1 \end{bmatrix}, \quad
    C = \begin{bmatrix} b_n & b_{n-1} & \dots & b_1 \end{bmatrix}
\end{equation*}

\subsection{2. Bentuk Kanonik Keterobservasian (\textit{Observable Canonical Form})}
\begin{equation*}
    A = \begin{bmatrix}
        -a_1 & 1 & 0 & \dots & 0 \\
        -a_2 & 0 & 1 & \dots & 0 \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        -a_{n-1} & 0 & 0 & \dots & 1 \\
        -a_n & 0 & 0 & \dots & 0
    \end{bmatrix}, \quad
    B = \begin{bmatrix} b_1 \\ b_2 \\ \vdots \\ b_n \end{bmatrix}, \quad
    C = \begin{bmatrix} 1 & 0 & \dots & 0 \end{bmatrix}
\end{equation*}

\subsection{3. Bentuk Kanonik Diagonal / Jordan}
Jika akar penyebut memiliki nilai berbeda $p_1, p_2, \dots, p_n$:
\begin{equation*}
    A = \operatorname{diag}(p_1, p_2, \dots, p_n), \quad B = \begin{bmatrix} 1 \\ 1 \\ \vdots \\ 1 \end{bmatrix}, \quad C = \begin{bmatrix} c_1 & c_2 & \dots & c_n \end{bmatrix}
\end{equation*}

\vspace{6pt}

\section{Pemodelan Sistem Multivariabel (SIMO, MISO, MIMO)}

\subsection{1. Sistem SIMO (\textit{Single-Input Multiple-Output})}
Fungsi transfer dinyatakan sebagai matriks kolom berukuran $p \times 1$:
\begin{equation*}
    \mathbf{G}(s) = \begin{bmatrix} g_1(s) \\ g_2(s) \\ \vdots \\ g_p(s) \end{bmatrix} = \frac{\boldsymbol{\beta}_1 s^{n-1} + \boldsymbol{\beta}_2 s^{n-2} + \dots + \boldsymbol{\beta}_n}{s^n + a_1 s^{n-1} + \dots + a_n} + \mathbf{d}
\end{equation*}
Realisasi kanonik keterkontrolan memiliki matriks keluaran $C = \begin{bmatrix} \boldsymbol{\beta}_1 & \boldsymbol{\beta}_2 & \dots & \boldsymbol{\beta}_n \end{bmatrix} \in \mathbb{R}^{p \times n}$.

\subsection{2. Sistem MISO (\textit{Multiple-Input Single-Output})}
Fungsi transfer dinyatakan sebagai matriks baris berukuran $1 \times m$:
\begin{equation*}
    \mathbf{G}(s) = \begin{bmatrix} g_1(s) & g_2(s) & \dots & g_m(s) \end{bmatrix} = \frac{\boldsymbol{\eta}_1 s^{n-1} + \boldsymbol{\eta}_2 s^{n-2} + \dots + \boldsymbol{\eta}_n}{s^n + a_1 s^{n-1} + \dots + a_n} + \mathbf{d}
\end{equation*}
Realisasi kanonik keterobservasian memiliki matriks masukan $B = \begin{bmatrix} \boldsymbol{\eta}_1^T & \boldsymbol{\eta}_2^T & \dots & \boldsymbol{\eta}_n^T \end{bmatrix}^T \in \mathbb{R}^{n \times m}$.

\subsection{3. Sistem MIMO (\textit{Multi-Input Multi-Output}) \& Realisasi Minimal Gilbert}
Melalui ekspansi pecahan parsial $G(s) = D + \sum_{i=1}^r \frac{W_i}{s - \lambda_i}$.\\
Faktorisasi matriks residu rank $k_i$: $W_i = C_i B_i$ dengan $C_i \in \mathbb{R}^{p \times k_i}$ dan $B_i \in \mathbb{R}^{k_i \times m}$. Realisasi minimal berorde $n = \sum k_i$:
\begin{equation*}
    A = \operatorname{diag}(\lambda_1 I_{k_1}, \dots, \lambda_r I_{k_r}), \quad B = \begin{bmatrix} B_1 \\ \vdots \\ B_r \end{bmatrix}, \quad C = \begin{bmatrix} C_1 & \dots & C_r \end{bmatrix}
\end{equation*}

\vspace{6pt}

\section{Solusi Domain Waktu \& Matriks Transisi Keadaan}

Solusi analitik persamaan keadaan kontinu adalah:
\begin{equation*}
    x(t) = e^{At} x(0) + \int_{0}^{t} e^{A(t - \tau)} B u(\tau) \, d\tau
\end{equation*}
Di mana matriks transisi keadaan diperoleh melalui invers Transformasi Laplace:
\begin{equation*}
    \Phi(t) = e^{At} = \mathcal{L}^{-1}\{(sI - A)^{-1}\}
\end{equation*}

\vspace{6pt}

\section{Contoh Soal dan Pembahasan}

\noindent\textbf{Contoh: Realisasi Sistem SIMO ke Bentuk Kanonik Keterkontrolan}\\
Diberikan fungsi transfer SIMO:
\begin{equation*}
    T(s) = \frac{1}{s^2 + 7s + 12} \begin{bmatrix} 3s + 13 \\ 3s + 11 \end{bmatrix}
\end{equation*}

\textit{Penyelesaian:}

Dari polinomial karakteristik pada penyebut, diperoleh $a_1 = 7$ dan $a_2 = 12$. Vektor polinomial pembilang dipisahkan berdasarkan derajat $s$:
\begin{equation*}
    \begin{bmatrix} 3s + 13 \\ 3s + 11 \end{bmatrix} = \begin{bmatrix} 3 \\ 3 \end{bmatrix} s + \begin{bmatrix} 13 \\ 11 \end{bmatrix}
\end{equation*}
Sehingga $\boldsymbol{\beta}_1 = \begin{bmatrix} 3 \\ 3 \end{bmatrix}$ dan $\boldsymbol{\beta}_2 = \begin{bmatrix} 13 \\ 11 \end{bmatrix}$.

Persamaan ruang keadaan dalam bentuk kanonik keterkontrolan adalah:
\begin{align*}
    \begin{bmatrix} \dot{x}_1 \\ \dot{x}_2 \end{bmatrix} &= \begin{bmatrix} 0 & 1 \\ -12 & -7 \end{bmatrix} \begin{bmatrix} x_1 \\ x_2 \end{bmatrix} + \begin{bmatrix} 0 \\ 1 \end{bmatrix} u \\
    \begin{bmatrix} y_1 \\ y_2 \end{bmatrix} &= \begin{bmatrix} 13 & 3 \\ 11 & 3 \end{bmatrix} \begin{bmatrix} x_1 \\ x_2 \end{bmatrix}
\end{align*}
