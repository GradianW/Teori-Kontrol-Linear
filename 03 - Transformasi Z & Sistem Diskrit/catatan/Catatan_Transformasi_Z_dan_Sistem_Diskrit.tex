\section{Pendahuluan Sistem Waktu Diskrit \& Pencuplikan}

Dalam implementasi sistem kendali modern, pengontrol umumnya direalisasikan menggunakan komputer digital, mikrokontroler, atau DSP (\textit{Digital Signal Processor}). Sinyal kontinu dari plant fisik harus diubah menjadi deret diskrit melalui proses pencuplikan (\textit{sampling}), dan sebaliknya keluaran komputer diubah kembali menjadi sinyal kontinu menggunakan rangkaian penahan (\textit{hold circuit}).

\subsection{Klasifikasi Sinyal}
\begin{itemize}
    \item \textbf{Sinyal Kontinu (Analog)}: Didefinisikan untuk setiap nilai waktu $t \in \mathbb{R}$ kontinu, dinotasikan $x(t)$.
    \item \textbf{Sinyal Diskrit (Discrete-Time)}: Didefinisikan hanya pada titik-titik waktu diskrit $t = kT$ dengan $k \in \mathbb{Z}$, dinotasikan $x(k)$ atau $x(kT)$, di mana $T$ adalah periode sampling.
    \item \textbf{Sinyal Digital}: Sinyal waktu diskrit yang amplitudonya telah dikuantisasi ke dalam level-level diskrit tertentu.
\end{itemize}

\subsection{Proses Pencuplikan Ideal \& Teorema Sampling Shannon}
Proses pencuplikan ideal memodelkan sinyal tercuplik $x^*(t)$ sebagai modulasi kereta impuls Dirac:
\begin{equation*}
    x^*(t) = \sum_{k=0}^{\infty} x(kT) \delta(t - kT)
\end{equation*}

\kotakdefinisi{Teorema Sampling Nyquist-Shannon:}{
Agar sinyal kontinu $x(t)$ dengan batas frekuensi tertinggi $\omega_{max}$ dapat direkonstruksi secara sempurna dari sinyal sampel $x(kT)$, frekuensi pencuplikan $\omega_s$ harus memenuhi:
\begin{equation*}
    \omega_s \ge 2 \omega_{max} \quad \left( \text{atau } f_s \ge 2 f_{max} \right)
\end{equation*}
Jika syarat ini dilanggar, timbul fenomena percampuran spektrum frekuensi yang disebut \textbf{aliasing}.
}

\subsection{Penahan Orde Nol (\textit{Zero-Order Hold} / ZOH)}
Untuk memegang nilai sinyal diskrit $u(kT)$ konstan selama satu interval periode sampling $T$:
\begin{equation*}
    u(t) = u(kT) \quad \text{untuk } kT \le t < (k+1)T
\end{equation*}
Fungsi transfer dari ZOH dalam domain Laplace adalah:
\begin{equation*}
    G_h(s) = \frac{1 - e^{-Ts}}{s}
\end{equation*}

\section{Definisi Transformasi Z}

\kotakdefinisi{Definisi Transformasi Z Satu-Sisi (Unilateral):}{
Diberikan deret waktu diskrit $x(k) = x(kT)$ untuk $k \ge 0$, Transformasi Z didefinisikan sebagai:
\begin{equation*}
    \mathcal{Z}\{x(k)\} = X(z) = \sum_{k=0}^{\infty} x(k) z^{-k}
\end{equation*}
Di mana $z$ adalah variabel kompleks yang berelasi dengan variabel Laplace $s$ melalui $z = e^{sT}$.
}

\subsection{Tabel Pasangan Transformasi Z Sinyal Dasar}
\begin{center}
\small
\begin{tabular}{|l|c|c|c|}
\hline
\textbf{Nama Sinyal} & $x(t), t \ge 0$ & $x(k), k \ge 0$ & $X(z) = \mathcal{Z}\{x(k)\}$ \\ \hline
\textbf{Impuls Satuan} & $\delta(t)$ & $\delta(k) = \begin{cases} 1, & k=0 \\ 0, & k \ne 0 \end{cases}$ & $1$ \\ \hline
\textbf{Tangga Satuan (Step)} & $1(t)$ & $1(k) = 1$ & $\dfrac{z}{z-1} = \dfrac{1}{1-z^{-1}}$ \\ \hline
\textbf{Ramp Satuan} & $t$ & $kT$ & $\dfrac{T z}{(z-1)^2}$ \\ \hline
\textbf{Polinomial $k$} & - & $k$ & $\dfrac{z}{(z-1)^2}$ \\ \hline
\textbf{Eksponensial} & $e^{-at}$ & $e^{-akT} = (e^{-aT})^k$ & $\dfrac{z}{z - e^{-aT}}$ \\ \hline
\textbf{Pangkat Basis $a$} & - & $a^k$ & $\dfrac{z}{z-a}$ \\ \hline
\textbf{Sinusoidal} & $\sin(\omega t)$ & $\sin(\omega k T)$ & $\dfrac{z \sin(\omega T)}{z^2 - 2z \cos(\omega T) + 1}$ \\ \hline
\textbf{Cosinusoidal} & $\cos(\omega t)$ & $\cos(\omega k T)$ & $\dfrac{z (z - \cos(\omega T))}{z^2 - 2z \cos(\omega T) + 1}$ \\ \hline
\end{tabular}
\end{center}

\vspace{6pt}

\section{Sifat-Sifat dan Teorema Transformasi Z}

\begin{enumerate}
    \item \textbf{Linearitas}:
    $\mathcal{Z}\{\alpha x_1(k) + \beta x_2(k)\} = \alpha X_1(z) + \beta X_2(z)$
    \item \textbf{Pergeseran Waktu (Time Shifting)}:
    \begin{itemize}
        \item \textbf{Delay}: $\mathcal{Z}\{x(k-m)\} = z^{-m} X(z)$
        \item \textbf{Advance}:
        \begin{align*}
            \mathcal{Z}\{x(k+1)\} &= z X(z) - z x(0) \\
            \mathcal{Z}\{x(k+2)\} &= z^2 X(z) - z^2 x(0) - z x(1) \\
            \mathcal{Z}\{x(k+3)\} &= z^3 X(z) - z^3 x(0) - z^2 x(1) - z x(2)
        \end{align*}
    \end{itemize}
    \item \textbf{Penskalaan Kompleks}:
    $\mathcal{Z}\{a^{-k} x(k)\} = X(az), \quad \mathcal{Z}\{e^{-akT} x(k)\} = X(z e^{aT})$
    \item \textbf{Diferensiasi Kompleks}:
    $\mathcal{Z}\{k \, x(k)\} = -z \frac{d}{dz} X(z)$
    \item \textbf{Teorema Nilai Awal \& Nilai Akhir}:
    \begin{equation*}
        x(0) = \lim_{z \to \infty} X(z), \quad \lim_{k \to \infty} x(k) = \lim_{z \to 1} (1 - z^{-1}) X(z) = \lim_{z \to 1} (z-1) X(z)
    \end{equation*}
\end{enumerate}

\vspace{6pt}

\section{Invers Transformasi Z}

Untuk memperoleh $x(k) = \mathcal{Z}^{-1}\{X(z)\}$:
\begin{enumerate}
    \item \textbf{Metode Ekspansi Pecahan Parsial}:
    Ekspansikan $\dfrac{X(z)}{z} = \sum_{i=1}^n \frac{c_i}{z - p_i} \implies X(z) = \sum_{i=1}^n c_i \frac{z}{z - p_i} \implies x(k) = \sum_{i=1}^n c_i (p_i)^k$.
    \item \textbf{Metode Pembagian Langsung (\textit{Power Series})}:
    Membagi pembilang dengan penyebut dalam pangkat menurun $z^{-1}$.
    \item \textbf{Metode Integral Inversi (Residu)}:
    $x(k) = \sum_i \lim_{z \to p_i} (z - p_i) X(z) z^{k-1}$.
\end{enumerate}

\vspace{6pt}

\section{Fungsi Pulsa Transfer (\textit{Pulse Transfer Function})}

\subsection{Persamaan Beda Linear Time-Invariant}
Diberikan persamaan beda:
\begin{equation*}
    y(k) + a_1 y(k-1) + \dots + a_n y(k-n) = b_0 u(k) + b_1 u(k-1) + \dots + b_m u(k-m)
\end{equation*}
Fungsi transfer pulsa adalah:
\begin{equation*}
    G(z) = \frac{Y(z)}{U(z)} = \frac{b_0 z^n + b_1 z^{n-1} + \dots + b_m z^{n-m}}{z^n + a_1 z^{n-1} + \dots + a_n}
\end{equation*}

\subsection{Diskritisasi Plant Kontinu dengan ZOH}
Jika suatu plant kontinu $G_p(s)$ didahului oleh penahan ZOH:
\begin{equation*}
    G(z) = (1 - z^{-1}) \mathcal{Z}\left\{ \frac{G_p(s)}{s} \right\}
\end{equation*}

\vspace{6pt}

\section{Representasi Ruang Keadaan Sistem Diskrit}

\kotakdefinisi{Persamaan Ruang Keadaan LTI Diskrit:}{
\begin{align*}
    x(k+1) &= G x(k) + H u(k) \quad &\text{(\textbf{Persamaan Keadaan Diskrit})} \\
    y(k) &= C x(k) + D u(k) \quad &\text{(\textbf{Persamaan Keluaran Diskrit})}
\end{align*}
}

\subsection{Diskritisasi dari Ruang Keadaan Kontinu}
Untuk sistem kontinu $\dot{x}(t) = A x(t) + B u(t)$, diskritisasi dengan ZOH pada periode sampling $T$:
\begin{equation*}
    G = e^{AT}, \quad H = \left( \int_{0}^{T} e^{A\tau} \, d\tau \right) B = A^{-1}(e^{AT} - I) B \quad (\text{jika } A \text{ nonsingular})
\end{equation*}

\subsection{Konversi Ruang Keadaan Diskrit ke Fungsi Transfer}
\begin{equation*}
    G(z) = \frac{Y(z)}{U(z)} = C (zI - G)^{-1} H + D = \frac{C \operatorname{adj}(zI - G) H}{\det(zI - G)} + D
\end{equation*}

\vspace{6pt}

\section{Bentuk-Bentuk Kanonik Sistem Diskrit}

Diberikan fungsi transfer pulsa $G(z) = \dfrac{b_1 z^{n-1} + \dots + b_n}{z^n + a_1 z^{n-1} + \dots + a_n}$:

\subsection{1. Bentuk Kanonik Keterkontrolan Diskrit}
\begin{equation*}
    G = \begin{bmatrix}
        0 & 1 & 0 & \dots & 0 \\
        0 & 0 & 1 & \dots & 0 \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & 0 & \dots & 1 \\
        -a_n & -a_{n-1} & -a_{n-2} & \dots & -a_1
    \end{bmatrix}, \quad
    H = \begin{bmatrix} 0 \\ 0 \\ \vdots \\ 0 \\ 1 \end{bmatrix}, \quad
    C = \begin{bmatrix} b_n & b_{n-1} & \dots & b_1 \end{bmatrix}
\end{equation*}

\subsection{2. Bentuk Kanonik Keterobservasian Diskrit}
\begin{equation*}
    G = \begin{bmatrix}
        -a_1 & 1 & 0 & \dots & 0 \\
        -a_2 & 0 & 1 & \dots & 0 \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        -a_{n-1} & 0 & 0 & \dots & 1 \\
        -a_n & 0 & 0 & \dots & 0
    \end{bmatrix}, \quad
    H = \begin{bmatrix} b_1 \\ b_2 \\ \vdots \\ b_n \end{bmatrix}, \quad
    C = \begin{bmatrix} 1 & 0 & \dots & 0 \end{bmatrix}
\end{equation*}

\vspace{6pt}

\section{Solusi Ruang Keadaan Diskrit \& Matriks Transisi Keadaan}

Solusi domain waktu secara rekursif:
\begin{equation*}
    x(k) = G^k x(0) + \sum_{j=0}^{k-1} G^{k-1-j} H u(j)
\end{equation*}
Di mana matriks transisi keadaan diskrit dihitung melalui:
\begin{equation*}
    \Psi(k) = G^k = \mathcal{Z}^{-1}\left\{ (zI - G)^{-1} z \right\}
\end{equation*}

\vspace{6pt}

\section{Kestabilan Sistem Waktu Diskrit}

\kotakdefinisi{Kriteria Kestabilan Sistem Diskrit:}{
Sistem LTI diskrit bersifat \textbf{stabil asimtotik} jika dan hanya jika seluruh kutub dari fungsi transfer $G(z)$ (atau seluruh nilai eigen dari matriks sistem $G$) terletak \textbf{di dalam lingkaran satuan} pada bidang kompleks $z$:
\begin{equation*}
    |\lambda_i| < 1, \quad \forall i = 1, 2, \dots, n
\end{equation*}
}

\subsection{Uji Kestabilan Jury Orde-2 (\texorpdfstring{$P(z) = a_0 z^2 + a_1 z + a_2 = 0, a_0 > 0$}{P(z)})}
\begin{enumerate}
    \item $|a_2| < a_0$
    \item $P(1) = a_0 + a_1 + a_2 > 0$
    \item $P(-1) = a_0 - a_1 + a_2 > 0$
\end{enumerate}

\section{Contoh Soal dan Pembahasan Komprehensif}

\noindent\textbf{Contoh 1: Solusi Persamaan Beda dengan Kondisi Awal}\\
Diberikan persamaan beda:
\begin{equation*}
    y(k+2) - 7y(k+1) + 12y(k) = 0, \quad y(0) = 0, \; y(1) = 1
\end{equation*}

\textit{Penyelesaian:}

Terapkan Transformasi Z dengan sifat pergeseran waktu maju:
\begin{align*}
    \mathcal{Z}\{y(k+2)\} &= z^2 Y(z) - z^2 y(0) - z y(1) = z^2 Y(z) - z \\
    \mathcal{Z}\{y(k+1)\} &= z Y(z) - z y(0) = z Y(z)
\end{align*}
Substitusi ke dalam persamaan beda:
\begin{align*}
    [z^2 Y(z) - z] - 7 [z Y(z)] + 12 Y(z) &= 0 \\
    (z^2 - 7z + 12) Y(z) &= z \\
    Y(z) &= \frac{z}{(z-3)(z-4)}
\end{align*}
Lakukan ekspansi pecahan parsial pada $\dfrac{Y(z)}{z}$:
\begin{align*}
    \frac{Y(z)}{z} &= \frac{1}{(z-3)(z-4)} = \frac{-1}{z-3} + \frac{1}{z-4} \\
    Y(z) &= -\frac{z}{z-3} + \frac{z}{z-4}
\end{align*}
Invers Transformasi Z dari masing-masing suku:
\begin{equation*}
    y(k) = -(3)^k + (4)^k = 4^k - 3^k, \quad k = 0, 1, 2, \dots
\end{equation*}

\vspace{10pt}

\noindent\textbf{Contoh 2: Diskritisasi Plant Kontinu dengan ZOH}\\
Diberikan plant kontinu $G_p(s) = \dfrac{2}{s + 2}$ dengan $T = 0.1$ detik. Tentukan fungsi transfer diskrit $G(z)$.

\textit{Penyelesaian:}
\begin{align*}
    G(z) &= (1 - z^{-1}) \mathcal{Z}\left\{ \frac{2}{s(s+2)} \right\} = \frac{z - 1}{z} \mathcal{Z}\left\{ \frac{1}{s} - \frac{1}{s+2} \right\} \\
    &= \frac{z - 1}{z} \left[ \frac{z(1 - e^{-2T})}{(z - 1)(z - e^{-2T})} \right] = \frac{1 - e^{-2T}}{z - e^{-2T}}
\end{align*}
Untuk $T = 0.1 \implies e^{-0.2} \approx 0.8187$:
\begin{equation*}
    G(z) = \frac{0.1813}{z - 0.8187}
\end{equation*}
